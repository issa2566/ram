  }

  /**
   * Check if table exists
   */
  static async tableExists() {
    try {
      const result = await pool.query(`
        SELECT EXISTS (
          SELECT FROM information_schema.tables 
          WHERE table_schema = 'public' 
          AND table_name = 'acha_products'
        )
      `);
      return result.rows[0].exists;
    } catch (error) {
      return false;
    }
  }

  /**
   * Ensure table exists before any operation
   */
  static async ensureTable() {
    const exists = await this.tableExists();
    if (!exists) {
      await this.initTable();
    }
  }

  /**
   * Find all acha products
   */
  static async findAll() {
    try {
      await this.ensureTable();
      const result = await pool.query(
        'SELECT * FROM acha_products ORDER BY created_at DESC'
      );
      return result.rows;
    } catch (error) {
      console.error('❌ Error in findAll:', error.message);
      return [];

    try {
      await this.ensureTable();
      const result = await pool.query(
        'SELECT * FROM acha_products WHERE id = $1',

      return result.rows[0] || null;

      description,

      images,
      quantity,
      product_references
    } = productData;

    const result = await pool.query(

       VALUES ($1, $2, $3, $4, $5, $6, $7)
       RETURNING *`,
      [
        sub_id,
        name || '',
        description || '',
        price || '0.000',
        images || [],
        quantity || 0,
        product_references || []
      ]
    );
    return result.rows[0];
  }

  /**
   * Update an acha product
   */
  static async update(id, productData) {
    await this.ensureTable();
    const updates = [];
    const values = [];
    let paramCount = 1;

    const fields = ['name', 'description', 'price', 'images', 'quantity', 'product_references'];

    fields.forEach(field => {
      if (productData[field] !== undefined) {
        updates.push(`${field} = $${paramCount++}`);
        values.push(productData[field]);
      }
    });

    if (updates.length === 0) {
      return await this.findById(id);
    }

    updates.push(`updated_at = CURRENT_TIMESTAMP`);
    values.push(id);

    const query = `UPDATE acha_products SET ${updates.join(', ')} WHERE id = $${paramCount} RETURNING *`;
    const result = await pool.query(query, values);
    return result.rows[0] || null;
  }

  /**
   * Delete an acha product
   */
  static async delete(id) {
    try {
      await this.ensureTable();
      const result = await pool.query(
        'DELETE FROM acha_products WHERE id = $1 RETURNING *',
        [id]
      );
      return result.rows[0] || null;
    } catch (error) {
      console.error('❌ Error in delete:', error.message);
      return null;
    }
  }

  /**
   * Decrease quantity by 1 (for vente hors ligne)
   */
  static async decreaseQuantity(id) {
    try {
      await this.ensureTable();
      const result = await pool.query(
        `UPDATE acha_products 
         SET quantity = GREATEST(0, quantity - 1), updated_at = CURRENT_TIMESTAMP 
         WHERE id = $1 
         RETURNING *`,
        [id]
      );
      return result.rows[0] || null;
    } catch (error) {
      console.error('❌ Error in decreaseQuantity:', error.message);
      return null;
    }
  }
}

      console.error('❌ Error saving to database:', error);
      // Revert UI change
      setProduct(previousProduct);
      
      toast({
        title: "Erreur",
        description: "Échec de la sauvegarde. Modification annulée.",
        variant: "destructive"
      });
    } finally {
      setIsSaving(false);
    }
  };
  
  // Handle quantity change (admin only)
  const handleQuantityChange = async (newQuantity: number) => {
    if (!isAdmin || !product?.id) return;
    
    const quantity = Math.max(0, newQuantity);
    
    // Optimistic update
    setProduct(prev => prev ? { ...prev, quantity } : null);
    
    await saveFieldToDatabase('quantity', quantity);
  };
  
  // Handle vente hors ligne (admin only)
  const handleVenteHorsLigne = async () => {
    if (!isAdmin || !product?.id || (product.quantity || 0) <= 0) return;
    
    setIsSaving(true);
    const previousQuantity = product.quantity;
    
    // Optimistic update
    setProduct(prev => prev ? { ...prev, quantity: Math.max(0, (prev.quantity || 0) - 1) } : null);
    
    try {
      const updatedProduct = await venteHorsLigneAchaProduct(product.id);
      setProduct(updatedProduct);
      
      toast({
        title: "Vente hors ligne",
        description: `Quantité mise à jour: ${updatedProduct.quantity}`,
      });
    } catch (error) {
      console.error('❌ Error performing vente hors ligne:', error);
      // Revert UI change
      setProduct(prev => prev ? { ...prev, quantity: previousQuantity } : null);
      
      toast({
        title: "Erreur",
        description: "Échec de la vente hors ligne",
        variant: "destructive"
      });
    } finally {
      setIsSaving(false);
    }
  };
  
  // Handle add reference (admin only)
  const handleAddReference = async () => {
    if (!isAdmin || !product?.id || !newReference.trim()) return;
    
    const currentRefs = product.product_references || [];
    const updatedRefs = [...currentRefs, newReference.trim()];
    
    // Optimistic update
    setProduct(prev => prev ? { ...prev, product_references: updatedRefs } : null);
    setNewReference("");
    
    await saveFieldToDatabase('product_references', updatedRefs);
  };
  
  // Handle remove reference (admin only)
  const handleRemoveReference = async (refToRemove: string) => {
    if (!isAdmin || !product?.id) return;
    
    const currentRefs = product.product_references || [];
    const updatedRefs = currentRefs.filter(ref => ref !== refToRemove);
    
    // Optimistic update
    setProduct(prev => prev ? { ...prev, product_references: updatedRefs } : null);
    
    await saveFieldToDatabase('product_references', updatedRefs);
  };
  
  // Image navigation handlers
  const handleNextImage = useCallback(() => {
    if (productData.images.length > 0) {
      setCurrentImageIndex((prev) => (prev + 1) % productData.images.length);
    }
  }, [productData.images.length]);
  
  const handlePrevImage = useCallback(() => {
    if (productData.images.length > 0) {
      setCurrentImageIndex((prev) => 
        (prev - 1 + productData.images.length) % productData.images.length
      );
    }
  }, [productData.images.length]);
  
  const handleDotClick = (index: number) => {
    setCurrentImageIndex(index);
  };
  
  // Touch gesture handlers for mobile swipe
  const handleTouchStart = (e: React.TouchEvent) => {
    touchStartX.current = e.touches[0].clientX;
  };
  
  const handleTouchMove = (e: React.TouchEvent) => {
    touchEndX.current = e.touches[0].clientX;
  };
  
  const handleTouchEnd = () => {
    if (!touchStartX.current || !touchEndX.current) return;
    const distance = touchStartX.current - touchEndX.current;
    const isLeftSwipe = distance > 50;
    const isRightSwipe = distance < -50;
    
    if (isLeftSwipe && productData.images.length > 1) {
      handleNextImage();
    }
    if (isRightSwipe && productData.images.length > 1) {
      handlePrevImage();
    }
  };
  
  // Keyboard navigation for images
  useEffect(() => {
    const handleKeyPress = (e: KeyboardEvent) => {
      if (productData.images.length > 1) {
        if (e.key === 'ArrowLeft') handlePrevImage();
        if (e.key === 'ArrowRight') handleNextImage();
      }
    };
    window.addEventListener('keydown', handleKeyPress);
    return () => window.removeEventListener('keydown', handleKeyPress);
  }, [handleNextImage, handlePrevImage, productData.images.length]);
  
  // Admin: Handle image upload
  const handleImageUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const files = e.target.files;
    if (!files || !product?.id) return;
    
    const remainingSlots = 5 - productData.images.length;
    const filesToProcess = Array.from(files).slice(0, remainingSlots);
    
    const newImages: string[] = [];
    
    for (const file of filesToProcess) {
      const reader = new FileReader();
      await new Promise<void>((resolve) => {
        reader.onloadend = () => {
          const base64 = reader.result as string;
          newImages.push(base64);
          resolve();
        };
        reader.readAsDataURL(file);
      });
    }
    
    const updatedImages = [...productData.images, ...newImages];
    setProductData(prev => ({ ...prev, images: updatedImages }));
    
    // Save to database
    await saveFieldToDatabase('images', updatedImages);
    
    // Reset input
    e.target.value = '';
  };
  
  // Admin: Delete image
  const handleDeleteImage = async (index: number) => {
    if (!product?.id) return;
    
    const newImages = productData.images.filter((_, i) => i !== index);
    setProductData(prev => ({ ...prev, images: newImages }));
    
    if (currentImageIndex >= newImages.length && newImages.length > 0) {
      setCurrentImageIndex(newImages.length - 1);
    } else if (newImages.length === 0) {
      setCurrentImageIndex(0);
    }
    
    // Save to database
    await saveFieldToDatabase('images', newImages);
  };
  
  // Admin: Save description
  const handleSaveDescription = async () => {
    if (!product?.id) return;
    
    setProductData(prev => ({ ...prev, description: tempDescription }));
    setIsEditingDescription(false);
    
    // Save to database
    await saveFieldToDatabase('description', tempDescription);
  };
  
  // Admin: Save price
  const handleSavePrice = async () => {
    if (!product?.id) return;
    
    setProductData(prev => ({ ...prev, price: tempPrice }));
    setIsEditingPrice(false);
    
    // Save to database
    await saveFieldToDatabase('price', tempPrice);
  };
  
  // Start editing description
  const startEditingDescription = () => {
    setTempDescription(productData.description);
    setIsEditingDescription(true);
  };
  
  // Start editing price
  const startEditingPrice = () => {
    setTempPrice(productData.price);
    setIsEditingPrice(true);
  };
  
  // Action handlers
  const handleAddToCart = () => {
    console.log("Adding to cart:", subId);
    toast({
      title: "Panier",
      description: "Produit ajouté au panier!",
    });
  };
  
  const handleOrder = () => {
    navigate('/cart');
  };
  
  const productTitle = decodeURIComponent(subId || "Produit");
  
  if (isLoading) {
    return (
      <div className="min-h-screen bg-gradient-to-b from-gray-50 via-white to-gray-50">
        <Header />
        <main className="max-w-6xl mx-auto px-4 sm:px-6 py-6 sm:py-10">
          <div className="flex items-center justify-center min-h-[400px]">
            <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-orange-500"></div>
          </div>
        </main>
        <Footer />
      </div>
    );
  }
  
  return (
    <div className="min-h-screen bg-gradient-to-b from-gray-50 via-white to-gray-50">
      <Header />
      
      <main className="max-w-6xl mx-auto px-4 sm:px-6 py-6 sm:py-10">
        {/* Breadcrumb */}
        <nav className="mb-6 text-sm text-gray-500 flex items-center gap-2">
          <a href="/" className="hover:text-orange-500 transition-colors">Accueil</a>
          <span>/</span>
          <span className="text-gray-900 font-medium">{productTitle}</span>
        </nav>
        
        <div className="grid grid-cols-1 lg:grid-cols-2 gap-8 lg:gap-12">
          {/* ========== IMAGE GALLERY SECTION ========== */}
          <section className="space-y-4">
            {/* Main Image Container */}
            <div 
              className="relative aspect-square bg-gradient-to-br from-gray-100 to-gray-50 rounded-2xl overflow-hidden shadow-lg group"
              onTouchStart={handleTouchStart}
              onTouchMove={handleTouchMove}
              onTouchEnd={handleTouchEnd}
            >
              {productData.images.length > 0 ? (
                <>
                  {/* Current Image */}
                  <div className="relative w-full h-full">
                    {productData.images.map((image, index) => (
                      <div
                        key={index}
                        className={`absolute inset-0 transition-opacity duration-500 ease-in-out ${
                          index === currentImageIndex ? 'opacity-100' : 'opacity-0'
                        }`}
                      >
                        <img
                          src={image}
                          alt={`${productTitle} - Image ${index + 1}`}
                          className="w-full h-full object-contain p-4"
                          loading={index === 0 ? "eager" : "lazy"}
                        />
                        
                        {/* Admin: Delete button on each image */}
                        {isAdmin && index === currentImageIndex && (
                          <button
                            onClick={() => handleDeleteImage(index)}
                            className="absolute top-3 right-3 bg-red-500 hover:bg-red-600 text-white p-2 rounded-full shadow-lg transition-all z-20"
                            title="Supprimer cette image"
                          >
                            <Trash2 className="w-4 h-4" />
                          </button>
                        )}
                      </div>
                    ))}
                  </div>
                  
                  {/* Navigation Arrows */}
                  {productData.images.length > 1 && (
                    <>
                      <button
                        onClick={handlePrevImage}
                        className="absolute left-3 top-1/2 -translate-y-1/2 bg-black/60 hover:bg-black/80 text-white p-2 sm:p-3 rounded-full opacity-100 sm:opacity-0 sm:group-hover:opacity-100 transition-all z-10 shadow-xl"
                        aria-label="Image précédente"
                      >
                        <ChevronLeft className="w-5 h-5" />
                      </button>
                      <button
                        onClick={handleNextImage}
                        className="absolute right-3 top-1/2 -translate-y-1/2 bg-black/60 hover:bg-black/80 text-white p-2 sm:p-3 rounded-full opacity-100 sm:opacity-0 sm:group-hover:opacity-100 transition-all z-10 shadow-xl"
                        aria-label="Image suivante"
                      >
                        <ChevronRight className="w-5 h-5" />
                      </button>
                    </>
                  )}
                  
                  {/* Dot Indicators */}
                  {productData.images.length > 1 && (
                    <div className="absolute bottom-4 left-1/2 -translate-x-1/2 flex gap-2 z-10">
                      {productData.images.map((_, index) => (
                        <button
                          key={index}
                          onClick={() => handleDotClick(index)}
                          className={`rounded-full transition-all duration-300 ${
                            index === currentImageIndex
                              ? 'bg-orange-500 w-8 h-3 shadow-lg'
                              : 'bg-white/70 hover:bg-white w-3 h-3'
                          }`}
                          aria-label={`Aller à l'image ${index + 1}`}
                        />
                      ))}
                    </div>
                  )}
                  
                  {/* Image Counter */}
                  <div className="absolute top-3 left-3 bg-black/60 text-white text-xs px-2 py-1 rounded-lg">
                    {currentImageIndex + 1} / {productData.images.length}
                  </div>
                </>
              ) : (
                /* Placeholder when no images */
                <div className="w-full h-full flex flex-col items-center justify-center text-gray-400">
                  <ImagePlus className="w-16 h-16 mb-4 opacity-50" />
                  <p className="text-sm">Aucune image disponible</p>
                  {isAdmin && (
                    <p className="text-xs mt-2 text-orange-500">Cliquez ci-dessous pour ajouter des images</p>
                  )}
                </div>
              )}
            </div>
            
            {/* Thumbnail Navigation */}
            {productData.images.length > 1 && (
              <div className="flex gap-2 justify-center overflow-x-auto pb-2">
                {productData.images.map((image, index) => (
                  <button
                    key={index}
                    onClick={() => handleDotClick(index)}
                    className={`flex-shrink-0 w-16 h-16 sm:w-20 sm:h-20 rounded-lg overflow-hidden border-2 transition-all ${
                      index === currentImageIndex
                        ? 'border-orange-500 ring-2 ring-orange-200 scale-105'
                        : 'border-gray-200 hover:border-gray-300'
                    }`}
                  >
                    <img
                      src={image}
                      alt={`Miniature ${index + 1}`}
                      className="w-full h-full object-cover"
                    />
                  </button>
                ))}
              </div>
            )}
            
            {/* Admin: Image Upload Section */}
            {isAdmin && productData.images.length < 5 && (
              <div className="bg-orange-50 border-2 border-dashed border-orange-300 rounded-xl p-4 text-center">
                <label className="cursor-pointer flex flex-col items-center gap-2">
                  <Upload className="w-8 h-8 text-orange-500" />
                  <span className="text-sm font-medium text-orange-600">
                    Ajouter des images ({productData.images.length}/5)
                  </span>
                  <span className="text-xs text-gray-500">
                    Cliquez pour télécharger (max 5 images)
                  </span>
                  <input
                    type="file"
                    accept="image/*"
                    multiple
                    onChange={handleImageUpload}
                    className="hidden"
                  />
                </label>
              </div>
            )}
          </section>
          
          {/* ========== PRODUCT INFO SECTION ========== */}
          <section className="space-y-6">
            {/* Product Title */}
            <div>
              <h1 className="text-2xl sm:text-3xl lg:text-4xl font-bold text-gray-900 mb-2">
                {productTitle}
              </h1>
              <div className="w-20 h-1 bg-gradient-to-r from-orange-500 to-orange-300 rounded-full" />
            </div>
            
            {/* ========== ADMIN-ONLY SECTION ========== */}
            {isAdmin && product && (
              <div className="space-y-4">
                {/* A) Quantité du produit & B) Vente Hors Ligne */}
                <div className="bg-purple-50 border border-purple-200 rounded-xl p-4">
                  <div className="flex items-center gap-2 mb-3">
                    <Package className="w-5 h-5 text-purple-600" />
                    <h3 className="font-semibold text-purple-800">Quantité du produit</h3>
                  </div>
                  
                  <div className="flex items-center gap-4 flex-wrap">
                    <div className="flex items-center gap-2">
                      <Input
                        type="number"
                        min="0"
                        value={product.quantity || 0}
                        onChange={(e) => handleQuantityChange(parseInt(e.target.value) || 0)}
                        className="w-24 text-center font-bold text-lg"
                        disabled={isSaving}
                      />
                      <span className="text-gray-600 text-sm">unités</span>
                    </div>
                    
                    <Button
                      onClick={handleVenteHorsLigne}
                      disabled={(product.quantity || 0) <= 0 || isSaving}
                      className="bg-purple-600 hover:bg-purple-700 text-white flex items-center gap-2"
                    >
                      <MinusCircle className="w-4 h-4" />
                      Vente hors ligne
                    </Button>
                  </div>
                  
                  {(product.quantity || 0) <= 0 && (
                    <p className="text-sm text-red-500 mt-2">Stock épuisé</p>
                  )}
                </div>
                
                {/* C) Références Section */}
                <div className="bg-green-50 border border-green-200 rounded-xl p-4">
                  <div className="flex items-center gap-2 mb-3">
                    <Tag className="w-5 h-5 text-green-600" />
                    <h3 className="font-semibold text-green-800">Références</h3>
                  </div>
                  
                  {/* Current References */}
                  <div className="flex flex-wrap gap-2 mb-3">
                    {(product.product_references || []).length === 0 ? (
                      <p className="text-sm text-gray-500 italic">Aucune référence</p>
                    ) : (
                      (product.product_references || []).map((ref, index) => (
                        <div
                          key={index}
                          className="flex items-center gap-1 bg-white border border-green-300 rounded-lg px-3 py-1.5 text-sm"
                        >
                          <span className="font-medium text-green-800">{ref}</span>
                          <button
                            onClick={() => handleRemoveReference(ref)}
                            className="text-red-500 hover:text-red-700 ml-1"
                            disabled={isSaving}
                            title="Supprimer cette référence"
                          >
                            <Trash2 className="w-3.5 h-3.5" />
                          </button>
                        </div>
                      ))
                    )}
                  </div>
                  
                  {/* Add New Reference */}
                  <div className="flex gap-2">
                    <Input
                      type="text"
                      placeholder="Nouvelle référence (ex: REF-001)"
                      value={newReference}
                      onChange={(e) => setNewReference(e.target.value)}
                      className="flex-1"
                      disabled={isSaving}
                      onKeyDown={(e) => {
                        if (e.key === 'Enter') {
                          e.preventDefault();
                          handleAddReference();
                        }
                      }}
                    />
                    <Button
                      onClick={handleAddReference}
                      disabled={!newReference.trim() || isSaving}
                      className="bg-green-600 hover:bg-green-700 text-white"
                    >
                      <Plus className="w-4 h-4" />
                    </Button>
                  </div>
                </div>
              </div>
            )}
            
            {/* Description Block */}
            <div className="bg-white rounded-xl border border-gray-200 shadow-sm overflow-hidden">
              <div className="flex items-center justify-between px-4 py-3 bg-gray-50 border-b border-gray-200">
                <h3 className="font-semibold text-lg text-gray-800">DESCRIPTION</h3>
                {isAdmin && !isEditingDescription && (
                  <button
                    onClick={startEditingDescription}
                    className="flex items-center gap-1.5 text-orange-500 hover:text-orange-600 text-sm font-medium transition-colors"
                  >
                    <Edit3 className="w-4 h-4" />
                    Modifier
                  </button>
                )}
              </div>
              <div className="p-4">
                {isEditingDescription ? (
                  <div className="space-y-3">
                    <Textarea
                      value={tempDescription}
                      onChange={(e) => setTempDescription(e.target.value)}
                      className="min-h-[120px] resize-none"
                      placeholder="Entrez la description du produit..."
                    />
                    <div className="flex gap-2 justify-end">
                      <Button
                        variant="outline"
                        size="sm"
                        onClick={() => setIsEditingDescription(false)}
                      >
                        <X className="w-4 h-4 mr-1" />
                        Annuler
                      </Button>
                      <Button
                        size="sm"
                        onClick={handleSaveDescription}
                        className="bg-orange-500 hover:bg-orange-600"
                        disabled={isSaving}
                      >
                        <Save className="w-4 h-4 mr-1" />
                        Sauvegarder
                      </Button>
                    </div>
                  </div>
                ) : (
                  <p className="text-gray-600 leading-relaxed whitespace-pre-wrap">
                    {productData.description}
                  </p>
                )}
              </div>
            </div>
            
            {/* Price Block */}
            <div className="bg-gradient-to-br from-orange-50 to-orange-100/50 rounded-xl border-2 border-orange-200 p-5">
              <div className="flex items-center justify-between mb-2">
                <h3 className="font-semibold text-lg text-gray-800">Prix</h3>
                {isAdmin && !isEditingPrice && (
                  <button
                    onClick={startEditingPrice}
                    className="flex items-center gap-1.5 text-orange-500 hover:text-orange-600 text-sm font-medium transition-colors"
                  >
                    <Edit3 className="w-4 h-4" />
                    Modifier
                  </button>
                )}
              </div>
              
              {isEditingPrice ? (
                <div className="space-y-3">
                  <div className="flex items-center gap-2">
                    <Input
                      type="text"
                      value={tempPrice}
                      onChange={(e) => setTempPrice(e.target.value)}
                      className="text-2xl font-bold max-w-[200px]"
                      placeholder="0.000"
                    />
                    <span className="text-xl font-bold text-gray-700">DT</span>
                  </div>
                  <div className="flex gap-2">
                    <Button
                      variant="outline"
                      size="sm"
                      onClick={() => setIsEditingPrice(false)}
                    >
                      <X className="w-4 h-4 mr-1" />
                      Annuler
                    </Button>
                    <Button
                      size="sm"
                      onClick={handleSavePrice}
                      className="bg-orange-500 hover:bg-orange-600"
                      disabled={isSaving}
                    >
                      <Save className="w-4 h-4 mr-1" />
                      Sauvegarder
                    </Button>
                  </div>
                </div>
              ) : (
                <p className="text-3xl sm:text-4xl font-bold text-orange-600">
                  {productData.price} <span className="text-xl text-gray-700">DT</span>
                </p>
              )}
            </div>
            
            {/* Action Buttons */}
            <div className="flex flex-col sm:flex-row gap-3 pt-2">
              <Button
                onClick={handleAddToCart}
                className="flex-1 bg-gradient-to-r from-orange-500 to-orange-600 hover:from-orange-600 hover:to-orange-700 text-white py-6 text-base sm:text-lg font-semibold shadow-lg hover:shadow-xl transition-all rounded-xl"
              >
                <ShoppingCart className="w-5 h-5 mr-2" />
                Ajouter au panier
              </Button>
              
              <Button
                onClick={handleOrder}
                className="flex-1 bg-gradient-to-r from-gray-900 to-gray-800 hover:from-gray-800 hover:to-gray-700 text-white py-6 text-base sm:text-lg font-semibold shadow-lg hover:shadow-xl transition-all rounded-xl"
              >
                <Zap className="w-5 h-5 mr-2" />
                Commander
              </Button>
            </div>
            
            {/* Admin Info Badge */}
            {isAdmin && (
              <div className="bg-blue-50 border border-blue-200 rounded-xl p-4 mt-4">
                <div className="flex items-start gap-3">
                  <div className="bg-blue-500 text-white p-2 rounded-lg">
                    <Edit3 className="w-4 h-4" />
                  </div>
                  <div>
                    <h4 className="font-semibold text-blue-800 text-sm">Mode Administrateur</h4>
                    <p className="text-xs text-blue-600 mt-1">
                      Vous pouvez modifier les images, la description, le prix, la quantité et les références de ce produit.
                      Les modifications sont sauvegardées automatiquement dans la base de données.
                    </p>
                  </div>
                </div>
              </div>
            )}
          </section>
        </div>
      </main>
      
      <Footer />
    </div>
  );
};

export default Acha;
